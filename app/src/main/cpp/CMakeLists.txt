cmake_minimum_required(VERSION 3.22.1)
project(screencopy_audio LANGUAGES C CXX)
set(OPUS_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/opus-1.5.2.tar.gz")
file(SHA256 "${OPUS_ARCHIVE}" OPUS_SHA256)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/opus-1.5.2.sha256" OPUS_EXPECTED)
string(STRIP "${OPUS_EXPECTED}" OPUS_EXPECTED)
if(NOT OPUS_SHA256 STREQUAL OPUS_EXPECTED)
    message(FATAL_ERROR "Pinned libopus source checksum mismatch")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/opus-1.5.2/CMakeLists.txt")
    file(ARCHIVE_EXTRACT INPUT "${OPUS_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
endif()
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_ENABLE_INTRINSICS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/opus-1.5.2" opus-build)
target_compile_options(opus PRIVATE -O2)
add_library(screencopy_opus SHARED opus_jni.cpp)
target_link_libraries(screencopy_opus PRIVATE opus)
target_compile_features(screencopy_opus PRIVATE cxx_std_17)
target_compile_options(screencopy_opus PRIVATE -Wall -Wextra -Werror)
target_link_options(screencopy_opus PRIVATE "-Wl,-z,max-page-size=16384")

option(SRT_TRIAL "Build the SRT transport trial" OFF)
if(SRT_TRIAL)
    set(SRT_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/srt-1.5.7.tar.gz")
    file(SHA256 "${SRT_ARCHIVE}" SRT_SHA256)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/srt-1.5.7.sha256" SRT_EXPECTED)
    string(STRIP "${SRT_EXPECTED}" SRT_EXPECTED)
    if(NOT SRT_SHA256 STREQUAL SRT_EXPECTED)
        message(FATAL_ERROR "Pinned libsrt source checksum mismatch")
    endif()
    if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/srt-1.5.7/.verified-v2-${SRT_SHA256}")
        file(ARCHIVE_EXTRACT INPUT "${SRT_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}"
            PATTERNS "srt-1.5.7/CMakeLists.txt" "srt-1.5.7/*.in" "srt-1.5.7/srtcore/*"
            "srt-1.5.7/common/*" "srt-1.5.7/scripts/*" "srt-1.5.7/haicrypt/*"
            "srt-1.5.7/LICENSE" "srt-1.5.7/*.cmake" "srt-1.5.7/cmake_object_lib_support.c")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/srt-1.5.7/.verified-v2-${SRT_SHA256}" "${SRT_SHA256}")
    endif()
    set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
    set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
    set(ENABLE_APPS OFF CACHE BOOL "" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(ENABLE_UNITTESTS OFF CACHE BOOL "" FORCE)
    set(ENABLE_ENCRYPTION OFF CACHE BOOL "" FORCE)
    set(ENABLE_CXX11 ON CACHE BOOL "" FORCE)
    add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/srt-1.5.7" srt-build)
    add_library(screencopy_srt SHARED ../srt/srt_transport.cpp)
    target_compile_features(screencopy_srt PRIVATE cxx_std_17)
    target_compile_options(screencopy_srt PRIVATE -Wall -Wextra -Werror)
    target_include_directories(screencopy_srt PRIVATE
        "${CMAKE_CURRENT_BINARY_DIR}/srt-1.5.7/srtcore" "${CMAKE_CURRENT_BINARY_DIR}/srt-build")
    target_link_libraries(screencopy_srt PRIVATE srt_static log)
    target_link_options(screencopy_srt PRIVATE "-Wl,-z,max-page-size=16384")
endif()
